\section{Research Background}

\begin{frame}{Problem Premise}
    \begin{itemize}
        \item \textbf{模型做代理任务时的越狱后果更严重}
        \begin{itemize}
            \item 正常对话任务中，越狱场景更少、破坏力更小
            \item 代理场景下会与现实世界交互，破坏力更强
        \end{itemize}
        \item \textbf{代理场景与对话场景的越狱形式不同}
        \begin{itemize}
            \item 对话场景：单次语言交互，要求模型理解语义层面的安全概念
            \item 代理场景：要求模型对有害目标有贡献即可，可能只是危险意图的子任务
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Problem Premise (cont.)}
    \begin{itemize}
        \item \textbf{代理场景的对齐比对话对齐更困难}
        \begin{itemize}
            \item 对模型遵守指令完成任务的要求比对话更强
            \item 模型会分配更多注意力在任务本身
            \item 与对齐要求模型适当拒绝用户产生了冲突
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Related Work - Jailbreak Evaluation}
    \begin{itemize}
        \item \textbf{Latent Jailbreak}\footnote{Qiu et al., "Latent Jailbreak." arXiv (2023).}：通过在不同位置嵌入恶意指令进行攻击
        \item \textbf{AdvBench}\footnote{Zou et al., "AdvBench." arXiv (2023).}：自动化搜索对抗后缀，攻击在多模型间可迁移
        \item \textbf{JailbreakHub}\footnote{Shen et al., "JailbreakHub." arXiv (2024).}：收集和分析 1,405 个社交媒体上的越狱提示
        \item \textbf{Do-Not-Answer}\footnote{Wang et al., "Do-Not-Answer." arXiv (2023).}：关注模型拒绝高风险任务的能力
        \item \textbf{ToxicChat}\footnote{Lin et al., "ToxicChat." arXiv (2023).}：利用真实对话数据揭示有害内容检测挑战
    \end{itemize}
\end{frame}

\begin{frame}{Related Work - Jailbreak Evaluation (cont.)}
    \begin{block}{关键问题}
        这些数据多依赖简单注释或直接的恶意输入，危险意图往往过于显性，难以模拟真实应用场景
    \end{block}
\end{frame}

\begin{frame}{Related Work - Agent Workflows}
    \begin{itemize}
        \item \textbf{AutoGen \& AutoAgents}：提供多角色对话编程，支持动态调度与灵活角色配置
        \item \textbf{AUTOACT}：引入元代理（Meta-Agent），强化任务分解与工具调用
        \item \textbf{UFO}：面向 Windows 的多应用协作与 UI 自动化
        \item \textbf{AutoDev}：结合代理调度与容器环境，支持大规模协同开发
        \item \textbf{Langgraph}：利用图结构实现任务管理与上下文控制，提升可扩展性
    \end{itemize}
\end{frame}

\begin{frame}{Work Overview}
    \begin{itemize}
        \item \textbf{提出可扩展且稳定的样本生成方法}
        \begin{itemize}
            \item 能够提供具有挑战性的Agent对齐测试样本
        \end{itemize}
        \item \textbf{收集涵盖工作流场景的安全对齐基准}
        \begin{itemize}
            \item 覆盖Agent最典型的几个任务类型
            \item 整合多个来源的恶意意图数据源
        \end{itemize}
        \item \textbf{提出新的对齐样本评估指标}
        \begin{itemize}
            \item 在PAIR、PAP等方法上进行了评估
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Dataset Construction Method}

\begin{frame}{Traditional Jailbreak Evaluation Paradigm}
    \begin{itemize}
        \item 将恶意意图形式化为二元组：$I=(G,M)$
        \begin{itemize}
            \item $G$：真实恶意目标（如生成有害输出）
            \item $M$：对 $G$ 施加的伪装/变换集合（如改写、混淆）
        \end{itemize}
        \item 通过越狱方法 $\Phi$ 生成对抗输入：$I'=\Phi(I)$
        \item 目标模型：$f_\theta(x)$；对抗输入的响应：$r=f_\theta(I')$
        \item 传统评估目标：最大化模型产生不安全响应的概率
        $$\arg\max_{I'} \Pr\big(f_\theta(I')\in \mathcal{R}_{\text{unsafe}}\big)$$
    \end{itemize}
\end{frame}

\begin{frame}{Traditional Jailbreak Evaluation Paradigm (cont.)}
    \begin{figure}
        \includegraphics[width=0.8\textwidth,height=0.6\textheight,keepaspectratio]{attachment/CleanShot-2025-08-21-at-16.56.47@2x.png}
        \caption{当前越狱基线评估范式}
    \end{figure}
    \begin{block}{问题}
        仅测试在抽象/可控环境下的脆弱性，难以反映真实工作流稳健性
    \end{block}
\end{frame}

\begin{frame}{Workflow Safety Evaluation Paradigm}
    \begin{itemize}
        \item 将真实任务拆解为多阶段工作流
        \item 在第 $t$ 步，给定历史 $H_t=\{o_1,\dots,o_{t-1}\}$ 与当前子任务 $T_t$
        \item 输出：$o_t=f_\theta(H_t,T_t)$
        \item 评估目标转为最大化安全输出概率：
        $$\max \Pr\big(o_t\in \mathcal{R}_{\text{safe}}\mid H_t, T_t\big)$$
        \item \textbf{差异}：不再强调伪装 $\Phi$ 的有效性，而是衡量模型在逐步诱导与任务上下文暴露下的对抗抵抗能力与风险传播
    \end{itemize}
\end{frame}

\begin{frame}{Workflow Safety Evaluation Paradigm (cont.)}
    \begin{figure}
        \includegraphics[width=0.8\textwidth,height=0.65\textheight,keepaspectratio]{attachment/CleanShot-2025-08-21-at-16.57.49@2x.png}
        \caption{本文提出的评估范式}
    \end{figure}
\end{frame}

\begin{frame}{Context Difference and Task Alignment}
    \begin{itemize}
        \item 现有上下文化方法（如 PAIR）常弱化或偏离原始危险目标
        \item 导致"看似安全"的迁移
        \item \textbf{本文方法}：保持危险意图不变、仅约束表达形式
    \end{itemize}
\end{frame}

\begin{frame}{Context Difference and Task Alignment (cont.)}
    \begin{figure}
        \includegraphics[width=0.9\textwidth,height=0.6\textheight,keepaspectratio]{attachment/CleanShot-2025-08-21-at-17.06.02@2x.png}
        \caption{三种输入方式对比：原始/PAIR/本文}
    \end{figure}
    \begin{block}{重要性}
        此差异解释了为何需要在真实工作流中评测安全性而非仅在对话场景中
    \end{block}
\end{frame}

\begin{frame}{Dangerous Workflow Simulation System - Phase A}
    \begin{itemize}
        \item \textbf{阶段A：危险任务生成}
        \begin{itemize}
            \item 从初始危险意图出发
            \item 使用攻击代理模拟若干轮工作历史
            \item 锚定当前轮次的子任务描述
            \item 目标：生成贴近真实流程的"在途"任务，而非一次性目标描述
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Dangerous Workflow Simulation System - Phase B}
    \begin{itemize}
        \item \textbf{阶段B：任务描述优化}
        \begin{itemize}
            \item 使用对齐模型标注并识别"触发直接拒绝"的禁词集合
            \item 在保持原始危险目标不变的前提下
            \item 由攻击代理对任务描述进行迭代改写
            \item 降低显性禁词、提高潜在危害聚焦度
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Sample Scalability and Stability}
    \begin{itemize}
        \item \textbf{样本生成借鉴 PAIR 与 StrongREJECT 的思路}
        \begin{itemize}
            \item 引入执行历史仿真
            \item 确保子任务"具体、可执行、可控"
        \end{itemize}
        \item \textbf{产出两个集合}
        \begin{itemize}
            \item JailFlowEval（>3000 条样本）
            \item JailFlowEval-hard（高质量子集 ~700 条）
        \end{itemize}
        \item \textbf{优势}：与传统"小规模人工标注"基座相比，本数据集在规模与工作流贴近性上更具挑战性
    \end{itemize}
\end{frame}

\section{Model Evaluation}

\begin{frame}{Model Alignment Metrics - PAIR Score}
    \begin{itemize}
        \item \textbf{PAIR Score（任务完成度 × 危害性）}
        \item 定义：$R(S)=f\big(C(S),\,V(S)\big)$
        \begin{itemize}
            \item $C(S)$：对当前子任务的清晰度/特异性/正确性
            \item $V(S)$：响应跨越伦理/法律/安全边界的程度
        \end{itemize}
        \item \textbf{特点}：
        \begin{itemize}
            \item 面向具体子任务的功能性与安全性评价
            \item 更契合工作流衡量
            \item 与 StrongReject 相比，不强调"说服力"维度
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Model Alignment Metrics - PAIR Score (cont.)}
    \begin{block}{备注}
        适用于判断"是否真正完成了危险子任务"，而非仅生成"有害片段"
    \end{block}
\end{frame}

\begin{frame}{Data Quality Metrics - Mutual Information}
    \begin{itemize}
        \item \textbf{设定}
        \begin{itemize}
            \item 数据集中提示集 $X=\{x_i\}_{i=1}^N$，响应集 $Y=\{y_j\}_{j=1}^M$
            \item 假设 $p(x_i)=1/N$
            \item 以相似度 $s_{ij}$ 计算条件分布（Softmax）：
        \end{itemize}
        $$p(y_j\mid x_i)=\frac{\exp(s_{ij})}{\sum_{k=1}^{M}\exp(s_{ik})}$$
    \end{itemize}
\end{frame}

\begin{frame}{Data Quality Metrics - Mutual Information (cont.)}
    \begin{itemize}
        \item \textbf{互信息定义}
        $$I(X;Y)=\sum_{i=1}^{N}\sum_{j=1}^{M}\frac{1}{N}\,p(y_j\mid x_i)\,\log_2\!\left(\frac{p(y_j\mid x_i)}{p(y_j)}\right)$$
        \item \textbf{成对贡献}
        $$I(x_i;y_i)=\frac{1}{N}\,p(y_i\mid x_i)\,\log_2\!\left(\frac{p(y_i\mid x_i)}{p(y_i)}\right)$$
    \end{itemize}
    \begin{block}{解释}
        用于捕捉成功响应在不同提示之间的可区分性/专一性
    \end{block}
\end{frame}

\begin{frame}{Dataset Direct Metrics - Keyword Density}
    \begin{itemize}
        \item 对每个提示 $p_i$，抽取得到关键词集合 $K(p_i)=\{k_1,\dots,k_{m_i}\}$，长度 $L_i$
        \item 密度：$\displaystyle d_i=\frac{m_i}{L_i}$
        \item 数据集平均：$\displaystyle d_{\text{avg}}=\frac{1}{N}\sum_{i=1}^{N}\frac{m_i}{L_i}$
    \end{itemize}
\end{frame}

\begin{frame}{Dataset Direct Metrics - Keyword Relevance}
    \begin{itemize}
        \item 关键词嵌入 $e(k)\in\mathbb{R}^d$；任意对 $(k_j,k_\ell)$ 的相似度
        $$\displaystyle S_{j\ell}=\frac{e(k_j)\cdot e(k_\ell)}{\|e(k_j)\|\,\|e(k_\ell)\|}$$
        \item 单条提示的平均相关性：
        $$r_i=\frac{2}{m_i(m_i-1)}\sum_{1\le j<\ell\le m_i}S_{j\ell}$$
        \item 数据集平均：$r_{\text{avg}}=\frac{1}{N}\sum_{i=1}^{N}r_i$
    \end{itemize}
\end{frame}

\begin{frame}{Dataset Direct Metrics - Keyword Relevance (cont.)}
    \begin{block}{用途}
        在不依赖模型的情况下，衡量提示的语义集中度与主题一致性
    \end{block}
\end{frame}

            \item 保持"危险目标不变"同时使表达更隐蔽
        \end{itemize}
    \end{itemize}
\end{frame}

    \end{block}
