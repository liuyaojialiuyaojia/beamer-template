\section{Introduction}

\begin{frame}{Industrial Agents Move to Workflows}
    \begin{itemize}
        \item Deployments now lean on service-oriented agents coordinating across organizations and edge-cloud systems.
        \item Agents plan, call tools, and iteratively revise using logged histories under governance limits.
        \item Safety stakes rise: orchestration can trigger real-world harm; \emph{OpenAI Biosafety Alert}\footnote{OpenAI. ``Autonomous Agents and Biosafety.'' 2024.} flagged the risk.
    \end{itemize}
\end{frame}

\begin{frame}{Limits of Chat-Centric Benchmarks}
    \begin{itemize}
        \item JailbreakHub, ToxicChat, and Do-Not-Answer remain dialogue-centric.
        \item They miss how risk propagates through service orchestration workflows.
        \item Refusal alone is a weak proxy for industrial agent safety.
    \end{itemize}
\end{frame}

\begin{frame}{Stage-Specific Risk}
    \begin{itemize}
        \item Unsafe plans plant failures for later stages.
        \item Unsafe executions can trigger irreversible real-world actions.
        \item Unsafe improvements may amplify harmful strategies.
    \end{itemize}
    \begin{block}{Takeaway}
        Refusing isolated prompts is insufficient; safety must be evaluated across the full workflow.
    \end{block}
\end{frame}

\section{Problem Statement}

\begin{frame}{Workflow-Centric Evaluation}
    \begin{itemize}
        \item Construct falsified jailbreak histories with explicit malicious intent.
        \item Evaluate each stage: Description $\rightarrow$ Planning $\rightarrow$ Execution $\rightarrow$ Reflection/Improvement.
        \item Goal: maximize $\Pr(o_t \in R_{\text{safe}} \mid H_t, T_t)$ under task pressure.
        \item Track how risk propagates between stages in realistic workflows.
    \end{itemize}
\end{frame}

\begin{frame}{Traditional Prompt Evaluation}
    \begin{itemize}
        \item Attacker directly supplies a dangerous goal and asks for completion.
        \item Single metric: whether the model refuses ($Response \in \{X,\checkmark\}$).
        \item Detached from agent context, so intent is obvious and risk propagation is ignored.
    \end{itemize}
\end{frame}

\section{Methodology}

\begin{frame}{Dangerous Workflow Simulation}
    \begin{itemize}
        \item Phase 1: Work History Simulation anchors an in-progress subtask around a dangerous mission.
        \item Goal: model a live workflow rather than cold-start malicious Q\&A.
        \item Phase 2: Safety Guard flags trigger words; Improver rewrites while keeping the dangerous goal intact.
        \item Result: prompts stay harmful but less obviously flagged by filters.
    \end{itemize}
\end{frame}

\begin{frame}{Simulation Pipeline}
    \begin{figure}

        \caption{Automated pipeline for dangerous subtask simulation.}

\begin{frame}{Data Distribution}
    \begin{itemize}
        \item Merged six corpora (StrongREJECT, AdvBench, Do-Not-Answer, ToxicChat, HarmBench, JailbreakBench) and filtered to 700 high-quality prompts.
        \item Eleven topics span malicious manipulation, harmful tutorials, evading liability, cybercrime, and more.
        \item Three key stages: Planning, Execution, and Improvement (error correction plus refinement).
    \end{itemize}
\end{frame}

\section{Evaluation Metrics}

\begin{frame}{PAIR Risk Score}
    \begin{itemize}
        \item $PAIR\_risk(S)=f\big(C(S),V(S)\big)$.
        \item $C(S)$: clarity, specificity, and correctness of task completion.
        \item $V(S)$: degree of violating ethical, legal, or safety boundaries.
        \item Low score means safer responses; high score shows harmful instructions executed well.
    \end{itemize}
\end{frame}

\begin{frame}{Mutual Information Across Stages}
    \begin{itemize}
        \item Measures how earlier dangerous prompts raise unsafe-output odds later.
        \item Pointwise MI: $PMI_{i}=\log_{2}\frac{p(y_{i}\mid x_{i})}{p(y_{i})}$.
        \item Captures risk propagation across planning, execution, and improvement stages.
    \end{itemize}
\end{frame}

\begin{frame}{Dataset Quality Metrics}
    \begin{itemize}
        \item Keyword density: $d_{i}=\frac{m_{i}}{L_{i}}$ measures how condensed dangerous cues are.
        \item Keyword correlation: cosine similarity between keyword embeddings reflects topic focus.
        \item Goal: concentrated semantics (high correlation) with natural phrasing (moderate density).
        \item Dataset-level averages ($d_{\text{avg}}, r_{\text{avg}}$) summarize prompt compactness and focus.
    \end{itemize}
\end{frame}

\begin{frame}{Key Findings: Open vs Closed}
    \begin{itemize}
        \item Open-source models show higher PAIR Risk and mutual information, meaning risk propagates more easily.
        \item DeepSeek V3 (deepseek-chat) carries the highest risk, highlighting tension between task competency and safety.
    \end{itemize}
\end{frame}

\begin{frame}{Key Findings: Closed Models \& Trade-offs}
    \begin{itemize}
        \item Claude 3.5 Sonnet and Haiku are most robust, consistently refusing dangerous workflows.
        \item GPT-4o mini has relatively higher risk among commercial models; Gemini 2.0 Flash sits in the middle.
        \item High task completion often coincides with unsafe behavior; refusal fails when danger is wrapped in workflows.
    \end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Key Contributions}
    \begin{itemize}
        \item JailFlowEval-Industrial: first workflow-centric agent safety benchmark with 700 high-quality prompts across 11 topics.
        \item Two-stage simulation pipeline keeps malicious goals explicit yet workflow-consistent.
        \item Metrics combine PAIR risk and stage-conditional mutual information to track risk propagation.
    \end{itemize}
\end{frame}

\begin{frame}{Audience Takeaways}
    \begin{itemize}
        \item Do not test only dialogue: safety regression should follow work histories and subtasks.
        \item Watch risk propagation: self-improvement loops can amplify vulnerabilities.
        \item Model choice matters: strong open-source models (e.g., DeepSeek V3) may be riskier than Claude for workflows.
    \end{itemize}
\end{frame}
